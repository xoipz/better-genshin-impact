## 为什么需要模块化

### 典型问题

当脚本超过500行时，你可能遇到这些问题：

```javascript
// ❌ 不好的做法：所有代码挤在一个文件中
// main.js (2000+ 行代码)

// 第1-200行：常量定义
const VERSION = "1.0.0";
const MAX_RETRIES = 3;
// ... 100个常量

// 第201-600行：工具函数
function sleep(ms) { ... }
function clickButton() { ... }
// ... 40个工具函数

// 第601-1200行：委托处理逻辑
function processCommission1() { ... }
function processCommission2() { ... }
// ... 30个委托处理函数

// 第1201-2000行：主流程
(async function() {
  // 主流程代码
})();

// 问题：
// - 难以找到特定功能的代码
// - 修改容易引入bug
// - 团队协作冲突频繁
// - 无法复用代码
```

### 模块化的好处

```javascript
// ✅ 好的做法：模块化架构

// main.js (100行)
eval(file.readTextSync("constants.js"));        // 常量
eval(file.readTextSync("lib/utils.js"));        // 工具函数
eval(file.readTextSync("lib/commissions.js"));  // 委托处理
eval(file.readTextSync("lib/core.js"));         // 核心逻辑

(async function() {
  await Core.executeMainProcess();
})();

// 好处：
// - 每个文件职责单一，易于维护
// - 可以独立测试每个模块
// - 多人可以同时编辑不同模块
// - 代码可以在不同项目中复用
```

## BGI模块化架构标准

### 核心规则

基于**08-JS脚本引擎架构与实现原理.md**的技术分析，BGI对模块化有严格要求：

```javascript
// ✅ BGI标准模块化结构

// 1. 顶层模块加载区域（必须在文件最开始）
eval(file.readTextSync("constants.js"));
eval(file.readTextSync("lib/utils.js"));
eval(file.readTextSync("lib/core.js"));

// 2. 函数定义区域（可选，如果需要在主流程外定义辅助函数）
function helperFunction() {
  // 辅助函数
}

// 3. BGI标准脚本入口（必须是文件的最后一个表达式）
(async function () {
  try {
    // 所有执行逻辑必须在此
    await Core.executeMainProcess();
    log.info("脚本执行完成");
  } catch (error) {
    log.error("脚本执行出错: {msg}", error.message);
    throw error;
  }
})();
```

### 关键技术要点

#### 1. 模块加载位置要求

```javascript
// ✅ 正确：在文件顶部加载模块
eval(file.readTextSync("lib/utils.js"));

(async function() {
  Utils.doSomething();  // ✅ 可以访问
})();

// ❌ 错误：在IIFE内部加载模块
(async function() {
  eval(file.readTextSync("lib/utils.js"));  // ❌ 作用域限制
  Utils.doSomething();  // 其他模块无法访问
})();
```

#### 2. 变量声明方式

```javascript
// === 在模块文件中 ===

// ✅ 使用 var 声明（BGI要求）
var Utils = {
  sleep: function(ms) { ... }
};

// ❌ 不要使用 const/let
const Utils = { ... };  // 在eval导入时可能有作用域问题
let helper = function() { ... };  // 同上
```

#### 3. IIFE位置要求

```javascript
// ✅ 正确：IIFE必须是文件的最后一个表达式
eval(file.readTextSync("lib/utils.js"));

(async function () {
  await doWork();
})();  // ✅ 返回Promise

// ❌ 错误：IIFE后面还有其他代码
(async function () {
  await doWork();
})();

var someVariable = 123;  // ❌ IIFE不是最后一个表达式
```

## 模块设计模式

### 1. 命名空间模式（推荐）

```javascript
// === lib/utils.js ===
var Utils = {
  // 延时函数
  sleep: function(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  },

  // 随机延时
  randomSleep: function(min, max) {
    var duration = Math.floor(Math.random() * (max - min + 1)) + min;
    return Utils.sleep(duration);
  },

  // 安全点击
  safeClick: async function(x, y) {
    await Utils.randomSleep(100, 300);
    click(x, y);
    await Utils.sleep(500);
  }
};

// === main.js ===
eval(file.readTextSync("lib/utils.js"));

(async function() {
  await Utils.safeClick(100, 200);  // ✅ 清晰的命名空间
})();
```

```javascript
  function createInstance() {
    var privateData = {
      maxRetries: 3,
      timeout: 5000
    };

    return {
      getMaxRetries: function() {
        return privateData.maxRetries;

        privateData.timeout = value;

        return privateData.timeout;
      }

    getInstance: function() {
      if (!instance) {
        instance = createInstance();
      }
      return instance;
    }
  };
```

### Step 3: 创建工具模块

```javascript
// === lib/utils.js ===
var Utils = {
  // 延时工具
  sleep: function(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  },

  // 重试机制
  retry: async function(fn, maxRetries, delay) {
    var retries = maxRetries || Constants.MAX_RETRIES;
    var retryDelay = delay || Constants.RETRY_DELAY;

    for (var i = 0; i < retries; i++) {
      try {
        return await fn();
      } catch (error) {
        log.warn("尝试 {current}/{max} 失败: {error}",
          i + 1, retries, error.message);

        if (i < retries - 1) {
          await Utils.sleep(retryDelay);
        } else {
          throw error;
        }
      }
    }
  },

  // OCR辅助
  findTextRegion: function(image, targetText) {
    var ocrResult = image.ocr();
    for (var i = 0; i < ocrResult.Regions.length; i++) {
      var region = ocrResult.Regions[i];
      if (region.Text.includes(targetText) && region.Score > 0.8) {
        return region;
      }
    }
    return null;
  }
};
```

```javascript
// === lib/core.js ===
var Core = {
  // 执行单次秘境
  runDomain: async function(domainName) {
    log.info("=== 开始执行秘境: {name} ===", domainName);

    try {
      // 1. 传送到秘境
      await Navigation.toDomain(domainName);

      // 2. 切换队伍

      await TeamManager.switchToTeam(domainConfig.team);

      // 3. 进入秘境
      await Navigation.enterDomain();

      // 4. 等待战斗完成
      await Combat.waitForCompletion();

      // 5. 领取奖励（简化）
      log.info("领取奖励");
      keyPress('VK_F');
      await Utils.sleep(2000);

      log.info("=== 秘境 {name} 完成 ===", domainName);
      return true;

    } catch (error) {
      log.error("秘境执行失败: {error}", error.message);
      throw error;
    }
  },

  // 主流程
  executeMainProcess: async function() {
    log.info("开始执行主流程");

    var domains = ["紫英庭", "太山府"];

    for (var i = 0; i < domains.length; i++) {
      await Core.runDomain(domains[i]);
      await Utils.sleep(2000);
    }

    log.info("主流程执行完成");
  }
};
```

```javascript
// === lib/state.js ===
var State = (function() {
  var state = {
    currentCommission: null,
    completedCount: 0,
    failedCount: 0,
    context: {}
  };

  return {
    get: function(key) {
      return state[key];
    },

    set: function(key, value) {
      state[key] = value;
    },

    update: function(updates) {
      for (var key in updates) {
        if (updates.hasOwnProperty(key)) {
          state[key] = updates[key];
        }
      }
    },

    getAll: function() {
      return Object.assign({}, state);
    }
  };
})();
```

### 2. 事件驱动通信

```javascript
// === lib/events.js ===
var EventBus = {
  events: {},

  on: function(eventName, callback) {
    if (!EventBus.events[eventName]) {
      EventBus.events[eventName] = [];
    }
    EventBus.events[eventName].push(callback);
  },

  emit: function(eventName, data) {
    var callbacks = EventBus.events[eventName];
    if (callbacks) {
      for (var i = 0; i < callbacks.length; i++) {
        callbacks[i](data);
      }
    }
  },

  off: function(eventName, callback) {
    var callbacks = EventBus.events[eventName];
    if (callbacks) {
      EventBus.events[eventName] = callbacks.filter(function(cb) {
        return cb !== callback;
      });
    }
  }
};

// === 使用示例 ===
// 监听器
EventBus.on("commission:completed", function(data) {
  log.info("委托完成: {name}", data.name);
  State.set("completedCount", State.get("completedCount") + 1);
});

// 触发事件
EventBus.emit("commission:completed", {
  name: "语言交流",
  duration: 120
});
```

## 常见陷阱和解决方案

### 陷阱1: 作用域问题

```javascript
// ❌ 错误：在IIFE内部eval
(async function() {
  eval(file.readTextSync("lib/utils.js"));
  Utils.doSomething();  // ✅ 这里可以用
})();

eval(file.readTextSync("lib/other.js"));
// 在other.js中调用Utils会失败，因为Utils只在IIFE作用域内

// ✅ 正确：在顶层eval
eval(file.readTextSync("lib/utils.js"));
eval(file.readTextSync("lib/other.js"));

(async function() {
  Utils.doSomething();  // ✅ 可以用
  Other.doSomething();  // ✅ 也可以用
})();
```

### 陷阱2: 循环依赖

```javascript
// ❌ 错误：模块A和模块B相互依赖

// lib/a.js
var A = {
  useB: function() {
    B.doSomething();  // 依赖B
  }
};

// lib/b.js
var B = {
  useA: function() {
    A.doSomething();  // 依赖A
  }
};

// ✅ 解决方案：引入中介模块

// lib/shared.js
var Shared = {
  commonFunction: function() {
    // 共同逻辑
  }
};

// lib/a.js
var A = {
  useShared: function() {
    Shared.commonFunction();
  }
};

// lib/b.js
var B = {
  useShared: function() {
    Shared.commonFunction();
  }
};
```

### 陷阱3: 全局变量污染

```javascript
// ❌ 错误：直接声明全局变量
var count = 0;  // 可能与其他脚本冲突

function increment() {
  count++;
}

// ✅ 正确：使用命名空间
var MyScript = {
  state: {
    count: 0
  },

  increment: function() {
    MyScript.state.count++;
  }
};
```

### 陷阱4: 异步处理不当

```javascript
// ❌ 错误：忘记await
async function processSteps(steps) {
  for (var i = 0; i < steps.length; i++) {
    doStep(steps[i]);  // ❌ 没有await，不会等待完成
  }
}

// ✅ 正确：正确使用await
async function processSteps(steps) {
  for (var i = 0; i < steps.length; i++) {
    await doStep(steps[i]);  // ✅ 等待每一步完成
  }
}
```

## 性能优化建议

### 1. 按需加载模块

```javascript
// ❌ 不推荐：加载所有模块
eval(file.readTextSync("lib/module1.js"));

// 其他模块在需要时再加载
```

### 2. 缓存计算结果

```javascript
// lib/cache.js
var Cache = {
  data: {},

  get: function(key, generator) {
    if (!Cache.data[key]) {
      Cache.data[key] = generator();
    }
    return Cache.data[key];
  },

  clear: function(key) {
    if (key) {
      delete Cache.data[key];
    } else {
      Cache.data = {};
    }
  }
};

// 使用示例
var expensiveResult = Cache.get("calculation", function() {
  // 耗时计算
  return doExpensiveCalculation();
});
```

### 3. 避免重复OCR

```javascript
var OCRCache = {
  cache: new Map(),

  getTextAt: function(region, cacheTime) {
    var key = region.X + "_" + region.Y + "_" + region.Width + "_" + region.Height;
    var now = Date.now();

    var cached = OCRCache.cache.get(key);
    if (cached && (now - cached.time < cacheTime)) {
      return cached.result;
    }

    var result = region.ocr();
    OCRCache.cache.set(key, { result: result, time: now });
    return result;
  }
};
```

## 最佳实践总结

### DO（推荐）

✅ **模块化分层清晰**
```javascript
// 明确的层次结构
constants.js -> 配置
lib/utils.js -> 工具
lib/core.js  -> 业务逻辑
main.js      -> 流程协调
```

✅ **使用命名空间**
```javascript
var MyModule = {
  function1: function() {},
  function2: function() {}
};
```

✅ **错误处理完善**
```javascript
try {
  await doSomething();
} catch (error) {
  log.error("操作失败: {msg}", error.message);
  throw error;
}
```

✅ **日志信息丰富**
```javascript
log.info("开始处理委托: {name}", commissionName);
log.info("步骤 {step}/{total} 完成", current, total);
```

### DON'T（避免）

❌ **不要在IIFE内部eval**
```javascript
// 错误
(async function() {
  eval(file.readTextSync("lib/utils.js"));
})();
```

❌ **不要使用全局变量**
```javascript
// 错误
var globalCount = 0;  // 容易冲突
```

❌ **不要忽略异步**
```javascript
// 错误
doAsyncWork();  // 忘记await
```

❌ **不要硬编码值**
```javascript
// 错误
sleep(5000);  // 应该用常量
// 正确
sleep(Constants.TIMEOUT);
```

## 实战案例：从单文件到模块化

### 重构前（单文件，600行）

```javascript
// old-main.js (600+ 行)
const VERSION = "1.0.0";
const MAX_RETRIES = 3;

function sleep(ms) { ... }
function click(x, y) { ... }
function processCommission1() { ... }
// ... 很多函数

(async function() {
  // ... 200行主流程代码
})();
```

### 重构后（模块化）

```javascript
// === constants.js (30行) ===
var Constants = {
  VERSION: "1.0.0",
  MAX_RETRIES: 3
};

// === lib/utils.js (50行) ===
var Utils = {
  sleep: function(ms) { ... },
  click: function(x, y) { ... }
};

// === lib/commissions.js (200行) ===
var Commissions = {
  process1: function() { ... },
  process2: function() { ... }
};

// === lib/core.js (150行) ===
var Core = {
  executeMainProcess: async function() { ... }
};

// === main.js (20行) ===
eval(file.readTextSync("constants.js"));
eval(file.readTextSync("lib/utils.js"));
eval(file.readTextSync("lib/commissions.js"));
eval(file.readTextSync("lib/core.js"));

(async function() {
  await Core.executeMainProcess();
})();
```

**重构收益**：
- ✅ 每个文件职责单一，易于理解
- ✅ 可以单独测试每个模块
- ✅ 多人可以同时编辑不同模块
- ✅ 代码可以在其他项目复用
