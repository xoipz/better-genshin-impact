### 脚本组项目管理

**代码位置**: `ScriptGroupProject.cs:15-120`

```javascript
// 创建脚本组项目
class ScriptGroupManager {
    constructor() {
        this.groups = new Map();
    }
    
    // 注册脚本组
    registerGroup(groupConfig) {
        log.info(`注册脚本组: ${groupConfig.name}`);
        this.groups.set(groupConfig.name, groupConfig);
    }
    
    // 执行脚本组
    async executeGroup(groupName) {
        const group = this.groups.get(groupName);
        if (!group) {
            throw new Error(`脚本组不存在: ${groupName}`);
        }
        
        log.info(`开始执行脚本组: ${groupName}`);
        
        if (group.executionOrder === 'sequential') {
            await this.executeSequential(group);
        } else {
            await this.executeParallel(group);
        }
    }
    
    // 顺序执行
    async executeSequential(group) {
        for (const script of group.scripts) {
            if (!script.enabled) continue;
            
            try {
                log.info(`执行脚本: ${script.name}`);
                await this.executeScript(script);
                log.info(`✓ ${script.name} 完成`);
            } catch (error) {
                log.error(`✗ ${script.name} 失败: ${error.message}`);
                
                if (group.errorHandling === 'stop') {
                    throw error;
                }
            }
        }
    }
    
    // 并行执行
    async executeParallel(group) {
        const promises = group.scripts
            .filter(script => script.enabled)
            .map(script => this.executeScript(script));
            
        const results = await Promise.allSettled(promises);
        
        results.forEach((result, index) => {
            const script = group.scripts[index];
            if (result.status === 'fulfilled') {
                log.info(`✓ ${script.name} 完成`);
            } else {
                log.error(`✗ ${script.name} 失败: ${result.reason.message}`);
            }
        });
    }
}
```

## settings 配置对象

### 概述

**代码位置**: `ScriptProject.cs:89`

`settings` 是一个动态配置对象，允许脚本开发者在 `manifest.json` 中定义可配置项。BGI会自动生成配置UI界面，用户设置的值会通过 `settings` 对象注入到JavaScript环境中，实现脚本的可配置化。

### 工作流程

1. **定义配置**: 在 `manifest.json` 中添加 `settings` 字段
2. **生成UI**: BGI根据配置自动生成设置界面
3. **用户配置**: 用户在UI中设置参数值
4. **注入对象**: 配置值通过 `settings` 对象注入到JS环境
5. **脚本访问**: JS脚本通过 `settings.xxx` 访问配置值

```javascript
// 自动秘境脚本主文件

async function runAutoDomain() {
    try {
        // 读取用户配置
        const domainCount = settings.domainCount;
        const useCondensedResin = settings.useCondensedResin;
        const notifyOnComplete = settings.notifyOnComplete;
        
        log.info(`开始自动秘境，次数: ${domainCount}`);
        log.info(`浓缩树脂: ${useCondensedResin ? '启用' : '禁用'}`);
        
        // 执行秘境任务
        for (let i = 0; i < domainCount; i++) {
            log.info(`第 ${i + 1}/${domainCount} 次秘境`);
            
            // 创建秘境任务参数
            const domainTask = new SoloTask();
            domainTask.Name = "AutoDomain";
            domainTask.Config = {
                useCondensedResin: useCondensedResin
            };
            
            // 执行任务
            await dispatcher.RunTask(domainTask);
            
            log.info(`第 ${i + 1} 次秘境完成`);
            
            // 等待一段时间
            if (i < domainCount - 1) {
                await sleep(2000);
            }
        }
        
        log.info("所有秘境完成!");
        
        // 根据配置发送通知
        if (notifyOnComplete) {
            notification.Send(`秘境任务完成，共完成 ${domainCount} 次`);
        }
        
    } catch (error) {
        log.error(`秘境任务失败: ${error.message}`);
        notification.Error(`秘境任务失败: ${error.message}`);
    }
}

// 执行主函数

})();
```

```javascript
// 配置验证工具函数
function validateSettings() {
    const errors = [];
    
    // 验证数字范围
    if (settings.domainCount < 1 || settings.domainCount > 10) {
        errors.push("秘境次数必须在1-10之间");
    }
    
    // 验证必需配置
    if (typeof settings.useCondensedResin === 'undefined') {
        errors.push("缺少浓缩树脂配置");
    }
    
    if (errors.length > 0) {
        log.error("配置验证失败:");
        errors.forEach(err => log.error(`- ${err}`));
        return false;
    }
    
    return true;
}

// 提供默认值
function getSettingWithDefault(key, defaultValue) {
    return settings[key] !== undefined ? settings[key] : defaultValue;
}

// 使用示例
const attackCount = getSettingWithDefault('attackCount', 5);
const enableDebug = getSettingWithDefault('enableDebug', false);
```

### 注意事项

1. **可选性**: `settings` 对象仅在脚本定义了配置项时才会注入
2. **类型安全**: 访问配置前建议进行类型检查
3. **默认值**: 始终在 `manifest.json` 中提供合理的默认值
4. **用户友好**: 配置项的 `label` 和 `description` 要清晰易懂
5. **验证**: 在使用配置值前进行必要的验证

## 委托处理流程配置

### process.json 流程配置文件

委托处理脚本使用`process.json`文件定义执行步骤流程，支持多种步骤类型的组合使用。

**文件位置**: `assets/process/{委托名}/{地点}/process.json`

### 步骤类型说明

#### 1. 地图追踪步骤
```json
{
  "type": "地图追踪", 
  "data": "追踪文件名.json",
  "note": "执行地图追踪到指定位置"
}
```

#### 2. 对话步骤
```json
{
  "type": "对话",
  "data": {
    "skipCount": 5,
    "priorityOptions": ["选项1", "选项2"],
    "npcWhiteList": ["NPC名称"]
  },
  "note": "执行对话交互"
}
```

#### 3. 委托描述检测步骤
```json
{
  "type": "委托描述检测",
  "data": "目标关键词",
  "note": "检测委托描述中的关键词",
  "run": "条件执行文件.json"
}
```

**关键特性**:
- **data字段支持字符串格式**: 直接写关键词字符串，代码会自动转换为对象格式
- **条件执行**: 当检测到关键词时，会执行`run`字段指定的文件
- **OCR识别**: 使用OCR技术识别屏幕上的委托描述文本
- **重试机制**: 支持最多3次检测重试

**实现原理**:
```javascript

    static async waitForUI(uiText, timeout = 10000) {
        const startTime = Date.now();

export function createRetryWrapper(maxRetries = 3) {
    return function(target, propertyKey, descriptor) {
        const originalMethod = descriptor.value;
        
        descriptor.value = async function(...args) {
            for (let attempt = 1; attempt <= maxRetries; attempt++) {
                try {
                    return await originalMethod.apply(this, args);
                } catch (error) {
                    if (attempt === maxRetries) throw error;
                    
                    log.warn(`重试 ${attempt}/${maxRetries}: ${error.message}`);
                    await sleep(1000 * attempt);
                }
            }
        };
        
        return descriptor;
    };
}
```

## 数据文件格式规范

### 数据文件格式要求

基于AutoCommission项目的技术发现，BGI脚本中的数据文件必须严格遵循代码期望的JSON格式：

**错误的数据格式（会导致运行时错误）**:
```json
{
  "commissions": [
    {"name": "委托名", "type": "talk"}
  ]
}
```

**正确的数据格式**:
```json
{
  "timestamp": "2025-07-25T12:41:00.000Z",
  "fightCommissions": [
    {
      "name": "攀高危险",
      "location": "苍风高地", 
      "country": "蒙德",
      "supported": true,
      "description": "清理史莱姆威胁"
    }
  ],
  "talkCommissions": [
    {
      "name": "餐品订单",
      "location": "蒙德城",
      "country": "蒙德", 
      "supported": true,
      "description": "配送食物订单"
    }
  ]
}
```

### 数据文件验证

在代码中使用数据时，建议添加格式验证：

```javascript
// 推荐的数据加载模式
function loadCommissionsData() {
  try {
    const data = JSON.parse(file.readTextSync(Constants.COMMISSIONS_DATA_PATH));
    
    // 验证必要字段存在
    if (!data.fightCommissions || !data.talkCommissions) {
      throw new Error('数据格式不正确：缺少fightCommissions或talkCommissions字段');
    }
    
    // 验证数据类型
    if (!Array.isArray(data.fightCommissions) || !Array.isArray(data.talkCommissions)) {
      throw new Error('数据格式不正确：fightCommissions和talkCommissions必须是数组');
    }
    
    return data;
  } catch (error) {
    log.error('数据文件加载失败:', error.message);
    return { fightCommissions: [], talkCommissions: [] };
  }
}
```

### 常见数据格式错误

1. **字段名称不匹配**: 使用`commissions`而非`fightCommissions`和`talkCommissions`
2. **数据类型错误**: 期望数组但提供对象
3. **缺少必要字段**: 缺少`supported`、`location`等必要属性
4. **JSON格式错误**: 语法错误导致解析失败
