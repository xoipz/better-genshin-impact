- **类型**: `number` (整数)
- **单位**: 毫秒 (milliseconds)

## 实用功能示例

### 判断每日重置时间

```javascript
/**
 * 检查是否已经过了每日4点重置
 * @returns {boolean} 如果已过4点返回true，否则false
 */
function isAfterDailyReset() {
    const offset = ServerTime.GetServerTimeZoneOffset();
    const serverTime = new Date(Date.now() + offset);

    const hour = serverTime.getHours();

    // 原神每日4点重置
    return hour >= 4;
}

// 使用示例
if (isAfterDailyReset()) {
    log.info("今日已重置，可以执行每日任务");
} else {
    log.info("尚未到每日重置时间");
}
```

### 计算距离下次重置的时间

```javascript
/**
 * 计算距离下次每日重置的剩余时间
 * @returns {object} 包含小时、分钟、秒的对象
 */
function getTimeUntilDailyReset() {
    const offset = ServerTime.GetServerTimeZoneOffset();
    const serverTime = new Date(Date.now() + offset);

    // 计算下次4点重置的时间
    const nextReset = new Date(serverTime);
    nextReset.setHours(4, 0, 0, 0);

    // 如果当前已经过了今天的4点，计算明天的4点
    if (serverTime.getHours() >= 4) {
        nextReset.setDate(nextReset.getDate() + 1);
    }

    // 计算时间差
    const diff = nextReset - serverTime;
    const hours = Math.floor(diff / (1000 * 60 * 60));
    const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));
    const seconds = Math.floor((diff % (1000 * 60)) / 1000);

    return { hours, minutes, seconds };
}

// 使用示例
const timeLeft = getTimeUntilDailyReset();
log.info(`距离每日重置还有: ${timeLeft.hours}小时 ${timeLeft.minutes}分钟 ${timeLeft.seconds}秒`);
```

### 判断是否在特定时间段内

```javascript
/**
 * 检查当前是否在指定的时间范围内
 * @param {number} startHour - 开始小时 (0-23)
 * @param {number} endHour - 结束小时 (0-23)
 * @returns {boolean}
 */
function isWithinTimeRange(startHour, endHour) {
    const offset = ServerTime.GetServerTimeZoneOffset();
    const serverTime = new Date(Date.now() + offset);
    const currentHour = serverTime.getHours();

    if (startHour <= endHour) {
        // 正常范围，如 6点到18点
        return currentHour >= startHour && currentHour < endHour;
    } else {
        // 跨天范围，如 22点到次日6点
        return currentHour >= startHour || currentHour < endHour;
    }
}

// 使用示例
if (isWithinTimeRange(6, 18)) {
    log.info("当前是白天时间（6:00-18:00）");
} else {
    log.info("当前是夜间时间");
}

if (isWithinTimeRange(22, 6)) {
    log.info("当前是深夜时段（22:00-6:00）");
}
```

### 活动时间判断

```javascript
/**
 * 判断指定活动是否在进行中
 * @param {string} startDate - 活动开始日期 "YYYY-MM-DD HH:mm:ss"
 * @param {string} endDate - 活动结束日期 "YYYY-MM-DD HH:mm:ss"
 * @returns {boolean}
 */
function isEventActive(startDate, endDate) {
    const offset = ServerTime.GetServerTimeZoneOffset();
    const serverTime = new Date(Date.now() + offset);

    const start = new Date(startDate);
    const end = new Date(endDate);

    return serverTime >= start && serverTime <= end;
}

// 使用示例
const eventStart = "2025-01-19 10:00:00";
const eventEnd = "2025-02-02 03:59:59";

if (isEventActive(eventStart, eventEnd)) {
    log.info("活动进行中，可以参与");

    // 计算活动剩余时间
    const offset = ServerTime.GetServerTimeZoneOffset();
    const serverTime = new Date(Date.now() + offset);
    const end = new Date(eventEnd);
    const remaining = end - serverTime;

    const days = Math.floor(remaining / (1000 * 60 * 60 * 24));
    const hours = Math.floor((remaining % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));

    log.info(`活动还剩: ${days}天 ${hours}小时`);
} else {
    log.info("活动未开始或已结束");
}
```

### 树脂恢复时间计算

```javascript
/**
 * 计算树脂恢复满所需时间
 * @param {number} currentResin - 当前树脂数量
 * @param {number} maxResin - 最大树脂容量（默认160）
 * @returns {object} 包含恢复时间的对象
 */
function calculateResinRecoveryTime(currentResin, maxResin = 160) {
    const resinToRecover = maxResin - currentResin;

    // 树脂恢复速度：8分钟1点
    const minutesNeeded = resinToRecover * 8;

    const offset = ServerTime.GetServerTimeZoneOffset();
    const serverTime = new Date(Date.now() + offset);
    const fullTime = new Date(serverTime.getTime() + minutesNeeded * 60 * 1000);

    const hours = Math.floor(minutesNeeded / 60);
    const minutes = minutesNeeded % 60;

    return {
        hours: hours,
        minutes: minutes,
        fullAt: fullTime,
        fullAtString: formatDateTime(fullTime)
    };
}

function formatDateTime(date) {
    const year = date.getFullYear();
    const month = String(date.getMonth() + 1).padStart(2, '0');
    const day = String(date.getDate()).padStart(2, '0');
    const hours = String(date.getHours()).padStart(2, '0');
    const minutes = String(date.getMinutes()).padStart(2, '0');

    return `${year}-${month}-${day} ${hours}:${minutes}`;
}

// 使用示例
const currentResin = 45;
const recovery = calculateResinRecoveryTime(currentResin);

log.info(`当前树脂: ${currentResin}/160`);
log.info(`恢复满需要: ${recovery.hours}小时 ${recovery.minutes}分钟`);
log.info(`预计恢复满时间: ${recovery.fullAtString}`);
```

### 定时任务调度

```javascript
/**
 * 检查是否到达指定的执行时间
 * @param {number} targetHour - 目标小时
 * @param {number} targetMinute - 目标分钟
 * @returns {boolean}
 */
function isScheduledTime(targetHour, targetMinute) {
    const offset = ServerTime.GetServerTimeZoneOffset();
    const serverTime = new Date(Date.now() + offset);

    const currentHour = serverTime.getHours();
    const currentMinute = serverTime.getMinutes();

    return currentHour === targetHour && currentMinute === targetMinute;
}

/**
 * 定时任务执行器
 */
async function scheduleTaskExecution() {
    const tasks = [
        { name: "每日委托", hour: 4, minute: 5 },
        { name: "领取纪行", hour: 4, minute: 10 },
        { name: "树脂消耗", hour: 20, minute: 0 }
    ];

    for (const task of tasks) {
        if (isScheduledTime(task.hour, task.minute)) {
            log.info(`执行定时任务: ${task.name}`);
            await executeTask(task.name);
        }
    }
}

async function executeTask(taskName) {
    switch (taskName) {

            log.info("开始执行每日委托...");

        case "树脂消耗":
            log.info("开始消耗树脂...");
            // 执行树脂消耗的逻辑
            break;
    }
}
```

## 时区说明

### 原神服务器时区

原神各服务器的时区如下：

- **国服**: UTC+8 (北京时间)
- **亚服**: UTC+8
- **美服**: UTC-5 (东部时间)
- **欧服**: UTC+1 (中欧时间)

BGI会自动检测并使用正确的服务器时区。

### 时区转换示例

```javascript
/**
 * 显示多个时区的时间
 */
function showMultipleTimezones() {
    const offset = ServerTime.GetServerTimeZoneOffset();
    const serverTime = new Date(Date.now() + offset);
    const localTime = new Date();

    log.info("=== 时间对比 ===");
    log.info(`本地时间: ${formatDateTime(localTime)}`);
    log.info(`服务器时间: ${formatDateTime(serverTime)}`);
    log.info(`时区偏移: ${offset / (1000 * 60 * 60)} 小时`);
}

function formatDateTime(date) {
    return date.toLocaleString('zh-CN', {
        year: 'numeric',
        month: '2-digit',
        day: '2-digit',
        hour: '2-digit',
        minute: '2-digit',
        second: '2-digit',
        hour12: false
    });
}

showMultipleTimezones();
```

### 2. 封装时间工具类

```javascript
/**
 * 服务器时间工具类
 */
class ServerTimeUtils {
    constructor() {
        this.offset = ServerTime.GetServerTimeZoneOffset();
    }

    // 获取服务器当前时间
    now() {
        return new Date(Date.now() + this.offset);
    }

    // 获取服务器当前小时
    getHour() {
        return this.now().getHours();
    }

    // 获取服务器当前星期几
    getDayOfWeek() {
        return this.now().getDay();
    }

    // 判断是否已过每日重置
    isAfterDailyReset() {
        return this.getHour() >= 4;
    }

    // 格式化服务器时间
    format(date = null) {
        const d = date || this.now();
        const year = d.getFullYear();
        const month = String(d.getMonth() + 1).padStart(2, '0');
        const day = String(d.getDate()).padStart(2, '0');
        const hours = String(d.getHours()).padStart(2, '0');
        const minutes = String(d.getMinutes()).padStart(2, '0');
        const seconds = String(d.getSeconds()).padStart(2, '0');

        return `${year}-${month}-${day} ${hours}:${minutes}:${seconds}`;
    }
}

// 使用示例
const serverTimeUtils = new ServerTimeUtils();
log.info("服务器时间:", serverTimeUtils.format());
log.info("当前小时:", serverTimeUtils.getHour());
log.info("已过重置:", serverTimeUtils.isAfterDailyReset());
```

## 注意事项

1. **时区偏移量是固定的**: 在脚本运行期间，时区偏移量不会改变，可以安全地缓存
2. **使用服务器时间判断**: 涉及游戏内重置、活动时间等判断时，应使用服务器时间而非本地时间
3. **考虑跨天情况**: 处理时间范围时要注意跨天的情况（如22:00-6:00）
4. **Date对象的月份**: JavaScript的Date对象月份从0开始（0-11），使用时需要+1
