## 基础异常处理

### try-catch基本用法

```javascript
async function basicErrorHandling() {
    try {
        // 可能出错的代码
        const position = genshin.getPositionFromMap();
        log.info(`当前位置: (${position.X}, ${position.Y})`);

    } catch (error) {
        // 捕获并处理错误
        log.error(`获取位置失败: ${error.message}`);

        // 可选：重新抛出错误
        // throw error;
    }
}
```

```javascript
async function asyncErrorHandling() {

        }

```

### finally块的使用

```javascript
async function resourceCleanup() {
    const hook = new KeyMouseHook();

    try {
        // 使用资源

            log.info(`按键: ${keyCode}`);
```

### 3. 重试异常

**RetryException** - 需要重试的异常

```javascript
async function handleRetryError() {
    const maxRetries = 3;

    for (let attempt = 1; attempt <= maxRetries; attempt++) {
        try {
            await performCriticalOperation();
            break; // 成功则退出循环

        } catch (error) {
            if (error.message.includes("Retry") && attempt < maxRetries) {
                log.warn(`操作失败，第 ${attempt} 次重试...`);
                await sleep(1000 * attempt); // 递增延迟
            } else {
                log.error(`操作最终失败: ${error.message}`);
                throw error;
            }
        }
    }
}
```

## 错误日志记录规范

### 日志级别选择

```javascript
function logErrorByLevel(error, context) {
    // 1. Debug级别 - 详细的调试信息
    log.debug(`[${context}] 错误堆栈: ${error.stack}`);

    // 2. Warn级别 - 可恢复的错误
    if (isRecoverableError(error)) {
        log.warn(`[${context}] 可恢复错误: ${error.message}`);
    }

    // 3. Error级别 - 严重错误
    if (isCriticalError(error)) {
        log.error(`[${context}] 严重错误: ${error.message}`);
    }
}

function isRecoverableError(error) {
    const recoverablePatterns = [
        "传送点未激活",
        "队伍设置失败",
        "Retry"
    ];
    return recoverablePatterns.some(pattern =>
        error.message.includes(pattern)
    );
}

function isCriticalError(error) {
    const criticalPatterns = [
        "无法连接",
        "权限不足",
        "文件不存在"
    ];
    return criticalPatterns.some(pattern =>
        error.message.includes(pattern)
    );
}
```

### 结构化错误日志

```javascript
class ErrorLogger {
    static logError(error, context = {}) {
        const errorInfo = {
            时间: new Date().toISOString(),
            消息: error.message,
            类型: error.name || 'Error',
            上下文: context,
            堆栈: error.stack
        };

        log.error(`错误详情: ${JSON.stringify(errorInfo, null, 2)}`);

        // 发送通知（仅严重错误）
        if (this.isCritical(error)) {
            notification.error(`严重错误: ${error.message}`);
        }
    }

    static isCritical(error) {
        return error.message.includes("Critical") ||
               error.message.includes("Fatal");
    }
}

// 使用示例
try {
    await performTask();
} catch (error) {
    ErrorLogger.logError(error, {
        任务名称: "自动采集",
        当前位置: genshin.getPositionFromMap(),
        尝试次数: 3
    });
}
```

## 异常恢复策略

### 1. 重试机制

```javascript
async function retryWithBackoff(operation, maxRetries = 3, baseDelay = 1000) {
    for (let attempt = 1; attempt <= maxRetries; attempt++) {
        try {
            return await operation();

        } catch (error) {
            const isLastAttempt = attempt === maxRetries;

            if (isLastAttempt) {
                log.error(`操作失败，已重试 ${maxRetries} 次: ${error.message}`);
                throw error;
            }

            // 指数退避延迟
            const delay = baseDelay * Math.pow(2, attempt - 1);
            log.warn(`操作失败，${delay}ms后进行第 ${attempt + 1} 次重试`);
            await sleep(delay);
        }
    }
}

// 使用示例
try {
    const result = await retryWithBackoff(async () => {
        return await genshin.tp(1000, 2000);
    }, 3, 1000);

    log.info("传送成功");
} catch (error) {
    log.error("传送最终失败");
}
```

### 2. 降级策略

```javascript
async function operationWithFallback() {
    try {
        // 尝试主要方法
        return await primaryMethod();

    } catch (error) {
        log.warn(`主要方法失败: ${error.message}，尝试备用方法`);

        try {
            // 尝试备用方法

        } catch (fallbackError) {
            log.error(`备用方法也失败: ${fallbackError.message}`);

            // 使用默认值

            return getDefaultValue();
        }
    }
}

}

async function fallbackMethod() {

}

function getDefaultValue() {
    // 默认值

}
```

### 3. 断点续传

```javascript
class TaskCheckpoint {
    constructor(checkpointFile) {
        this.file = checkpointFile;
        this.checkpoint = this.load();
    }

    load() {
        try {
            const content = file.readTextSync(this.file);
            return JSON.parse(content);
        } catch (error) {
            log.info("没有找到检查点文件，从头开始");
            return { completedTasks: [], lastPosition: null };
        }
    }

    save() {
        try {
            file.writeTextSync(
                this.file,
                JSON.stringify(this.checkpoint, null, 2)
            );
            log.debug("检查点已保存");
        } catch (error) {
            log.warn(`保存检查点失败: ${error.message}`);
        }
    }

    markCompleted(taskId) {
        this.checkpoint.completedTasks.push(taskId);
        this.save();
    }

    isCompleted(taskId) {
        return this.checkpoint.completedTasks.includes(taskId);
    }

    setPosition(position) {
        this.checkpoint.lastPosition = position;
        this.save();
    }
}

// 使用示例
async function resumableTask() {
    const checkpoint = new TaskCheckpoint("checkpoint.json");
    const tasks = ["任务1", "任务2", "任务3", "任务4"];

    try {
        for (const task of tasks) {
            if (checkpoint.isCompleted(task)) {
                log.info(`跳过已完成的任务: ${task}`);
                continue;
            }

            log.info(`执行任务: ${task}`);
            await executeTask(task);

            checkpoint.markCompleted(task);
            log.info(`任务完成: ${task}`);
        }

        log.info("所有任务完成");

    } catch (error) {
        log.error(`任务中断: ${error.message}`);
        log.info("下次运行将从中断处继续");
        throw error;
    }
}
```

## 常见错误场景

### 1. 文件操作错误

```javascript
async function safeFileOperation() {
    try {
        // 检查文件是否存在
        if (!file.isFolder("data")) {
            log.warn("数据目录不存在，使用默认配置");
            return getDefaultConfig();
        }

        const content = await file.readText("data/config.json");
        return JSON.parse(content);

    } catch (error) {
        if (error.message.includes("找不到文件")) {
            log.warn("配置文件不存在，创建默认配置");
            const defaultConfig = getDefaultConfig();
            file.writeTextSync("data/config.json",
                JSON.stringify(defaultConfig, null, 2));
            return defaultConfig;
        } else {
            log.error(`文件操作失败: ${error.message}`);
            throw error;
        }
    }
}
```

### 2. 网络请求错误

```javascript
async function safeHttpRequest(url) {
    try {
        const response = await http.Request("GET", url);

        if (response.status_code !== 200) {
            throw new Error(`HTTP错误: ${response.status_code}`);
        }

        return JSON.parse(response.body);

    } catch (error) {
        if (error.message.includes("UnauthorizedAccess")) {
            log.error("HTTP权限不足，请检查manifest.json配置");
            notification.error("HTTP请求被拒绝，请检查权限配置");
        } else if (error.message.includes("timeout")) {
            log.warn("HTTP请求超时，稍后重试");
        } else {
            log.error(`HTTP请求失败: ${error.message}`);
        }
        throw error;
    }
}
```

### 脚本级错误处理

```javascript
async function mainScript() {
    try {
        log.info("=== 脚本开始执行 ===");

        // 初始化

    }
}

(async function() {
    try {
        await mainScript();
    } catch (error) {
        log.error("脚本异常终止");
        // 错误已记录，不再抛出
    }
```

## 最佳实践

### 1. 错误处理原则

- ✅ **捕获具体异常**: 根据异常类型采取不同措施
- ✅ **记录详细信息**: 包含上下文、堆栈、时间戳
- ✅ **优雅降级**: 提供备用方案和默认值
- ✅ **资源清理**: 使用finally确保资源释放
- ✅ **用户通知**: 严重错误通知用户

### 2. 不要做的事

- ❌ **空catch块**: 不要忽略错误
- ❌ **过度捕获**: 不要捕获所有错误
- ❌ **隐藏错误**: 不要吞掉重要错误
- ❌ **重复日志**: 避免同一错误多次记录

### 3. 错误处理检查清单

```javascript
// ✅ 良好的错误处理示例
async function goodErrorHandling() {
    const resource = null;

    try {
        // 1. 获取资源
        resource = await acquireResource();

        // 2. 使用资源
        await useResource(resource);

        // 3. 记录成功
        log.info("操作成功");

    } catch (error) {
        // 4. 记录详细错误
        log.error(`操作失败: ${error.message}`);

        // 5. 根据错误类型处理
        if (isRecoverableError(error)) {
            await handleRecoverableError(error);
        } else {
            // 6. 通知用户
            notification.error(`严重错误: ${error.message}`);
            throw error;
        }

    } finally {
        // 7. 清理资源
        if (resource) {
            await releaseResource(resource);
        }
    }
}
```
