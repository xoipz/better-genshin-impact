#### 2. OCR识别数量
```javascript
async function readPrimogems() {
  var gameImage = captureGameRegion();

  // 定义识别区域（根据1080p坐标）
  var primogemRegion = {
    x: 1600,
    y: 200,
    width: 200,
    height: 50
  };

  // 裁剪区域
  var croppedImage = gameImage.crop(primogemRegion);

  // OCR识别
  var ocrResult = croppedImage.ocr();
  var text = ocrResult.Text;

  // 提取数字（正则表达式）
  var numbers = text.match(/\d+/g);
  if (numbers && numbers.length > 0) {
    return parseInt(numbers[0]);
  }

  return 0;
}
```

#### 3. 数据持久化
```javascript

  var content = JSON.stringify(data, null, 2);
  file.writeTextSync("data/gacha_log.txt", content);
```

### 学习要点

✅ **适合学习**：

⚠️ **注意事项**：

- 数字提取要处理逗号分隔符（如"1,234"）

```javascript
var combatIconTemplate = file.readImageMatSync("assets/combat_icon.png");

async function detectCombatStatus() {
  while (true) {
    var gameImage = captureGameRegion();

    // 模板匹配
    var result = gameImage.find({
```

#### 3. 商店购买逻辑
```javascript
// 购买食材
async function buyItems(itemList) {
  log.info("开始购买食材");

  for (var i = 0; i < itemList.length; i++) {
    var itemName = itemList[i];
    log.info("购买: {item}", itemName);

    try {
      // 1. OCR识别物品名称
      var found = await findItemByOCR(itemName);

      if (found) {
        // 2. 点击物品
        click(found.x, found.y);
        await sleep(500);

        // 3. 最大化购买
        await clickMaxButton();
        await sleep(300);

        // 4. 确认购买
        await clickConfirmButton();
        await sleep(800);

        log.info("购买成功: {item}", itemName);
      } else {
        log.warn("未找到物品: {item}", itemName);
      }
    } catch (error) {
      log.error("购买失败 {item}: {error}", itemName, error.message);
    }
  }
}

// OCR识别物品
async function findItemByOCR(itemName) {
  var gameImage = captureGameRegion();

  // 定义商店区域
  var shopRegion = gameImage.crop({
    x: 800,
    y: 200,
    width: 1000,
    height: 800
  });

  // OCR识别
  var ocrResult = shopRegion.ocr();

  // 查找匹配的文本
  for (var i = 0; i < ocrResult.Regions.length; i++) {
    var region = ocrResult.Regions[i];
    if (region.Text.includes(itemName) && region.Score > 0.7) {
      return {
        x: region.Rect.Center.X,
        y: region.Rect.Center.Y
      };
    }
  }

  return null;
}
```

#### 4. 异常处理
```javascript
// 检测购买限制
async function checkPurchaseLimit() {
  var gameImage = captureGameRegion();
  var ocrResult = gameImage.ocr();

  // 检测"已达购买上限"文本
  for (var i = 0; i < ocrResult.Regions.length; i++) {
    var text = ocrResult.Regions[i].Text;
    if (text.includes("购买上限") || text.includes("库存不足")) {
      log.warn("检测到购买限制");
      return true;
    }
  }

  return false;
}
```

```javascript
var waypoints = [];
var isRecording = true;

async function recordLoop() {
  log.info("开始录制路径");

  while (isRecording) {
    try {
      // 获取当前位置
      var pos = await getCurrentPosition();

      // 添加航点
      waypoints.push({
        x: Math.round(pos.x),
        y: Math.round(pos.y),
        timestamp: new Date().toISOString()

      log.info("记录航点 #{index}: ({x}, {y})",

      // 检测是否打开大地图

        log.info("检测到大地图打开，停止录制");
        isRecording = false;
      }

    } catch (error) {
      log.error("录制出错: {error}", error.message);
    }
  }

  log.info("录制完成，共 {count} 个航点", waypoints.length);
}
```

```javascript
// 检测大地图是否打开
async function isBigMapOpen() {
```

```javascript
var AutoFightStep = {
  name: "auto-fight",

  // 执行步骤
  execute: async function(config, context) {
    log.info("开始自动战斗");

    // 1. 切换战斗队伍
    if (config.team) {
      await genshin.switchParty(config.team);
    }

  validate: function(config) {

      throw new Error("战斗超时时间不能少于30秒");
    }
    return true;
  }
};

StepRegistry.register(AutoFightStep);
```

#### 5. 执行引擎

```javascript
// === lib/execute.js ===
var ExecuteEngine = {
  // 执行流程
  executeProcess: async function(processFile, context) {
    // 1. 读取流程定义
    var processData = JSON.parse(file.readTextSync(processFile));
    log.info("执行流程: {name}", processData.name);

    // 2. 遍历步骤
    for (var i = 0; i < processData.steps.length; i++) {
      var step = processData.steps[i];
      log.info("步骤 {current}/{total}: {type}",
        i + 1, processData.steps.length, step.type);

      try {
        // 3. 获取步骤处理器
        var processor = StepRegistry.get(step.type);
        if (!processor) {
          throw new Error("未找到处理器: " + step.type);
        }

        // 4. 验证配置
        if (processor.validate) {
          processor.validate(step.config);
        }

        // 5. 执行步骤
        await processor.execute(step.config, context);

        log.info("步骤完成: {type}", step.type);

      } catch (error) {
        log.error("步骤失败 {type}: {error}",
          step.type, error.message);

        // 错误处理策略
        if (step.required) {
          throw error;  // 必需步骤失败则终止
        } else {
          log.warn("跳过可选步骤");
          continue;
        }
      }
    }

    log.info("流程完成: {name}", processData.name);
  }
};
```
